## 一、需求概述

### 1.1 目标

将建模参数暴露到前端，让用户可手动调整：
- **默认值**从 `config.py` 读取
- 用户调整后**点击"应用"**才生效
- 参数**保存到后端**（与 session_id 绑定）+ **localStorage**
- 参数变化时**缓存失效**
- UI 形式：左侧控制面板的**折叠区域**
- 输入形式：**混合**（数值用滑块+输入框，枚举用下拉，N_LAYERS 用滑块+自动开关）
- 前端 + 后端**双重校验**参数范围

### 1.2 参数清单

**A 组（插值参数，全部暴露）**：

| 参数 | 类型 | 范围 | 默认值来源 |
|:---|:---|:---|:---|
| `RBF_FUNCTION` | 枚举 | linear / thin_plate / multiquadric / inverse / gaussian | config.py |
| `RBF_SMOOTH` | 数值 | 0.001 ~ 10 | config.py |
| `GRID_RESOLUTION` | 数值 | 10 ~ 100 | config.py |
| `N_LAYERS` | 数值/自动 | 5 ~ 50 或 None | config.py |

**B 组（钻孔样式，只暴露两个）**：

| 参数 | 类型 | 范围 | 默认值来源 |
|:---|:---|:---|:---|
| `DRILL_STYLE` | 枚举 | cylinder / line | config.py |
| `DRILL_RADIUS` | 数值 | 0.5 ~ 20 | config.py |

**不暴露的参数**：`PROJECTION_ZONE`、`DRILL_CYLINDER_SIDES`、`DRILL_SAMPLE_STEP`、`VIEW_CONFIGS` 等。

## 三、新建 `params_manager.py`

```python
"""
用户参数管理：管理每个会话的建模参数。
参数与 session_id 绑定，保存在 uploads/{session_id}/user_params.json 中。
"""
import os
import json
import hashlib

from config import UPLOAD_FOLDER, SESSION_PARAMS_FILENAME, USER_ADJUSTABLE_PARAMS


def get_default_params():
    """获取所有可调参数的默认值"""
    defaults = {}
    for key, meta in USER_ADJUSTABLE_PARAMS.items():
        defaults[key] = meta['default']
    return defaults


def get_params_path(session_id):
    """获取参数文件路径"""
    return os.path.join(UPLOAD_FOLDER, session_id, SESSION_PARAMS_FILENAME)


def load_session_params(session_id):
    """
    加载会话参数。
    如果参数文件不存在，返回默认参数。
    """
    if not session_id:
        return get_default_params()
    
    params_path = get_params_path(session_id)
    if not os.path.exists(params_path):
        return get_default_params()
    
    try:
        with open(params_path, 'r', encoding='utf-8') as f:
            saved = json.load(f)
        # 与默认值合并，确保新增的参数有默认值
        defaults = get_default_params()
        defaults.update(saved)
        return defaults
    except Exception as e:
        print(f"!!! 读取参数失败: {e}")
        return get_default_params()


def save_session_params(session_id, params):
    """
    保存会话参数。只保存可调参数，其他键忽略。
    """
    if not session_id:
        return False, "缺少 session_id"
    
    # 过滤：只保留可调参数
    filtered = {}
    for key in USER_ADJUSTABLE_PARAMS.keys():
        if key in params:
            filtered[key] = params[key]
    
    # 校验
    ok, err = validate_params(filtered)
    if not ok:
        return False, err
    
    # 保存
    params_path = get_params_path(session_id)
    os.makedirs(os.path.dirname(params_path), exist_ok=True)
    
    try:
        with open(params_path, 'w', encoding='utf-8') as f:
            json.dump(filtered, f, ensure_ascii=False, indent=2)
        return True, None
    except Exception as e:
        return False, str(e)


def validate_params(params):
    """
    校验参数范围。
    返回: (是否合法, 错误信息)
    """
    for key, value in params.items():
        if key not in USER_ADJUSTABLE_PARAMS:
            continue
        meta = USER_ADJUSTABLE_PARAMS[key]
        ptype = meta['type']
        
        # 枚举校验
        if ptype == 'enum':
            if value not in meta['options']:
                return False, f"{key} 的值 '{value}' 不在允许的选项中: {meta['options']}"
        
        # 数值校验
        elif ptype == 'number':
            try:
                v = float(value)
            except (TypeError, ValueError):
                return False, f"{key} 必须是数字"
            if v < meta['min'] or v > meta['max']:
                return False, f"{key} 的值 {v} 超出范围 [{meta['min']}, {meta['max']}]"
        
        # 数值或自动校验
        elif ptype == 'number_or_auto':
            if value is None:
                continue
            try:
                v = int(value)
            except (TypeError, ValueError):
                return False, f"{key} 必须是整数或 null"
            if v < meta['min'] or v > meta['max']:
                return False, f"{key} 的值 {v} 超出范围 [{meta['min']}, {meta['max']}]"
    
    return True, None


def compute_params_hash(params):
    """
    计算参数哈希，用于缓存判断。
    """
    filtered = {}
    for key in sorted(USER_ADJUSTABLE_PARAMS.keys()):
        if key in params:
            filtered[key] = params[key]
    # 转为 JSON 字符串后哈希
    s = json.dumps(filtered, sort_keys=True, ensure_ascii=False)
    return hashlib.md5(s.encode('utf-8')).hexdigest()


def get_frontend_meta():
    """
    返回给前端的参数元信息（用于渲染 UI）。
    """
    meta = {}
    for key, info in USER_ADJUSTABLE_PARAMS.items():
        meta[key] = {
            'label': info['label'],
            'type': info['type'],
            'default': info['default'],
            'description': info.get('description', '')
        }
        if 'options' in info:
            meta[key]['options'] = info['options']
        if 'min' in info:
            meta[key]['min'] = info['min']
        if 'max' in info:
            meta[key]['max'] = info['max']
        if 'step' in info:
            meta[key]['step'] = info['step']
    return meta
```

def get_cached_result(session_id, task_type, input_hash, params_hash=None):
    """
    查询缓存。
    
    参数:
        session_id: 会话 ID
        task_type: 'model' 或 'ai_model'
        input_hash: 输入文件的哈希值
        params_hash: 参数哈希（可选）
    
    返回:
        命中: {'result': {...}, 'created_at': '...'}
        未命中: None
    """
    meta = load_cache_meta(session_id)
    entry = meta.get(task_type)
    if not entry:
        return None
    
    # 检查输入文件哈希
    if entry.get('input_hash') != input_hash:
        return None
    
    # 检查参数哈希（如果提供）
    if params_hash is not None:
        if entry.get('params_hash') != params_hash:
            return None
    
    # 检查是否超时
    created_at = datetime.fromisoformat(entry['created_at'])
    if datetime.now() - created_at > timedelta(hours=CACHE_TTL_HOURS):
        return None
    
    # 检查结果文件是否还存在
    result_dir = entry.get('result_dir')
    if not result_dir or not os.path.exists(result_dir):
        return None
    
    return {
        'result': entry.get('result', {}),
        'created_at': entry['created_at']
    }


def save_cached_result(session_id, task_type, input_hash, result, result_dir, params_hash=None):
    """
    保存缓存。
    """
    meta = load_cache_meta(session_id)
    meta[task_type] = {
        'input_hash': input_hash,
        'params_hash': params_hash,
        'created_at': datetime.now().isoformat(),
        'result': result,
        'result_dir': result_dir
    }
    save_cache_meta(session_id, meta)
```

    """
    当参数变化时，清除该 session 的模型和 AI 模型缓存。
    """
    meta = load_cache_meta(session_id)

@app.route('/api/save_params', methods=['POST'])
def save_params_api():
    """
    保存当前会话的参数。
    如果参数与旧参数不同，清除模型缓存。
    """
    data = request.json
    session_id = data.get('session_id')
    params = data.get('params', {})
    
    if not session_id:
        return jsonify({'error': '缺少 session_id'}), 400
    
    from params_manager import (
        load_session_params, save_session_params,
        compute_params_hash
    )
    from cache_manager import clear_cache_by_params_change
    
    # 读取旧参数
    old_params = load_session_params(session_id)
    old_hash = compute_params_hash(old_params)
    
    # 保存新参数
    ok, err = save_session_params(session_id, params)
    if not ok:
        return jsonify({'error': err}), 400
    
    # 计算新哈希
    new_params = load_session_params(session_id)
    new_hash = compute_params_hash(new_params)
    
    # 如果参数变化，清除缓存
    if old_hash != new_hash:
        clear_cache_by_params_change(session_id)
        log_and_print(session_id, f"==> 参数已变更，模型缓存已清除")
    
    return jsonify({
        'success': True,
        'params': new_params,
        'params_hash': new_hash
    })
```

### 8.1 在文件顶部新增参数管理逻辑

在 DOM 元素定义之后添加：

// ==================== 高级参数管理 ====================
let currentParams = {};
let currentParamsMeta = {};
let savedParamsHash = '';

// 加载参数元信息和当前值
async function loadSessionParams() {
    if (!sessionId) return;
    
    try {
        const response = await fetch(`/api/session_params?session_id=${sessionId}`);
        const data = await response.json();
        
        if (data.success) {
            currentParamsMeta = data.meta || {};
            savedParamsHash = data.params_hash || '';
            
            // 先从 localStorage 读取（优先级更高）
            const localSaved = localStorage.getItem(`params_${sessionId}`);
            let params = data.params || {};
            
            if (localSaved) {
                try {
                    const parsed = JSON.parse(localSaved);
                    params = { ...params, ...parsed };
                } catch (e) {}
            }
            
            currentParams = params;
            renderParamsUI();
        }
    } catch (error) {
        console.error('加载参数失败:', error);
    }
}

// 渲染参数 UI
function renderParamsUI() {
    const container = document.getElementById('paramsContent');
    const loading = document.getElementById('paramsLoading');
    if (!container || !loading) return;
    
    loading.style.display = 'none';
    container.style.display = 'block';
    
    let html = '';
    for (const [key, meta] of Object.entries(currentParamsMeta)) {
        const value = currentParams[key] !== undefined ? currentParams[key] : meta.default;
        html += renderParamItem(key, meta, value);
    }
    container.innerHTML = html;
    
    // 绑定事件
    for (const key of Object.keys(currentParamsMeta)) {
        const meta = currentParamsMeta[key];
        if (meta.type === 'number') {
            const slider = document.getElementById(`param_${key}_slider`);
            const input = document.getElementById(`param_${key}_input`);
            if (slider && input) {
                slider.addEventListener('input', () => {
                    input.value = slider.value;
                });
                input.addEventListener('input', () => {
                    slider.value = input.value;
                });
            }
        } else if (meta.type === 'number_or_auto') {
            const slider = document.getElementById(`param_${key}_slider`);
            const input = document.getElementById(`param_${key}_input`);
            const autoCheck = document.getElementById(`param_${key}_auto`);
            if (slider && input && autoCheck) {
                autoCheck.addEventListener('change', () => {
                    const isAuto = autoCheck.checked;
                    slider.disabled = isAuto;
                    input.disabled = isAuto;
                });
                // 初始化
                if (value === null || value === undefined) {
                    autoCheck.checked = true;
                    slider.disabled = true;
                    input.disabled = true;
                }
            }
        }
    }
}

// 渲染单个参数项
function renderParamItem(key, meta, value) {
    let html = `<div class="param-item">`;
    html += `<label class="param-label" title="${meta.description || ''}">${meta.label}</label>`;
    
    if (meta.type === 'enum') {
        html += `<select id="param_${key}" class="param-select">`;
        for (const opt of meta.options) {
            const selected = (opt === value) ? 'selected' : '';
            html += `<option value="${opt}" ${selected}>${opt}</option>`;
        }
        html += `</select>`;
    } else if (meta.type === 'number') {
        html += `<div class="param-number-row">`;
        html += `<input type="range" id="param_${key}_slider" 
                 min="${meta.min}" max="${meta.max}" step="${meta.step}" value="${value}"
                 class="param-slider">`;
        html += `<input type="number" id="param_${key}_input" 
                 min="${meta.min}" max="${meta.max}" step="${meta.step}" value="${value}"
                 class="param-input">`;
        html += `</div>`;
    } else if (meta.type === 'number_or_auto') {
        html += `<div class="param-auto-row">`;
        html += `<label class="param-auto-label"><input type="checkbox" id="param_${key}_auto"> 自动</label>`;
        html += `</div>`;
        html += `<div class="param-number-row">`;
        html += `<input type="range" id="param_${key}_slider" 
                 min="${meta.min}" max="${meta.max}" step="${meta.step}" value="${value || meta.min}"
                 class="param-slider">`;
        html += `<input type="number" id="param_${key}_input" 
                 min="${meta.min}" max="${meta.max}" step="${meta.step}" value="${value || meta.min}"
                 class="param-input">`;
        html += `</div>`;
    }
    
    if (meta.description) {
        html += `<div class="param-description">${meta.description}</div>`;
    }
    
    html += `</div>`;
    return html;
}

// 收集用户当前调整的参数
function collectParamsFromUI() {
    const params = {};
    for (const [key, meta] of Object.entries(currentParamsMeta)) {
        if (meta.type === 'enum') {
            const el = document.getElementById(`param_${key}`);
            if (el) params[key] = el.value;
        } else if (meta.type === 'number') {
            const el = document.getElementById(`param_${key}_input`);
            if (el) params[key] = parseFloat(el.value);
        } else if (meta.type === 'number_or_auto') {
            const autoCheck = document.getElementById(`param_${key}_auto`);
            if (autoCheck && autoCheck.checked) {
                params[key] = null;  // 自动模式
            } else {
                const el = document.getElementById(`param_${key}_input`);
                if (el) params[key] = parseInt(el.value);
            }
        }
    }
    return params;
}

// 应用参数
async function applyParams() {
    if (!sessionId) {
        setStatus('!!! 请先上传文件', 'error');
        return;
    }
    
    const params = collectParamsFromUI();
    const hint = document.getElementById('paramsHint');
    
    try {
        const response = await fetch('/api/save_params', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ session_id: sessionId, params: params })
        });
        const data = await response.json();
        
        if (data.error) {
            setStatus('!!! 参数保存失败: ' + data.error, 'error');
            if (hint) hint.textContent = '保存失败: ' + data.error;
            return;
        }
        
        currentParams = data.params;
        savedParamsHash = data.params_hash;
        
        // 保存到 localStorage
        localStorage.setItem(`params_${sessionId}`, JSON.stringify(params));
        
        setStatus('==> 参数已应用，下次生成时生效', 'success');
        if (hint) {
            hint.textContent = '参数已应用，下次生成时生效';
            hint.style.color = '#2E7D32';
        }
        
        // 收起面板
        setTimeout(() => {
            document.getElementById('paramsBody').style.display = 'none';
            document.getElementById('paramsToggle').textContent = '▶';
        }, 800);
        
    } catch (error) {
        setStatus('!!! 参数保存失败: ' + error.message, 'error');
    }
}

// 绑定折叠面板事件
function initParamsPanel() {
    const header = document.getElementById('paramsHeader');
    const body = document.getElementById('paramsBody');
    const toggle = document.getElementById('paramsToggle');
    const applyBtn = document.getElementById('btnApplyParams');
    
    if (!header || !body || !toggle) return;
    
    header.addEventListener('click', () => {
        const isHidden = body.style.display === 'none';
        body.style.display = isHidden ? 'block' : 'none';
        toggle.textContent = isHidden ? '▼' : '▶';
        
        // 首次展开时加载参数
        if (isHidden && Object.keys(currentParamsMeta).length === 0 && sessionId) {
            loadSessionParams();
        }
    });
    
    if (applyBtn) {
        applyBtn.addEventListener('click', applyParams);
    }
}

// 初始化（在 DOMContentLoaded 中调用）
document.addEventListener('DOMContentLoaded', () => {
    initParamsPanel();
});
