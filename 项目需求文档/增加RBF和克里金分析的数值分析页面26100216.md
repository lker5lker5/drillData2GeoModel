# 第一部分：克里金数值分析

## 1.1 需求概述

### 1.1.1 核心目标

在主建模流程（RBF 插值）之外，新增一个独立的"克里金数值分析"功能，用于**佐证和对比 RBF 建模的准确性**。

**核心产出**：
1. 克里金插值结果（2D 平面，与 RBF 同一层对比）
2. 预测方差热力图（克里金的独特价值，量化不确定性）
3. RBF vs 克里金对比图（并排展示）
4. 可用于报告的分析结论

### 1.1.2 关键决策

| 项 | 决策 |
|:---|:---|
| 分析对象 | 所有岩性层（每个岩性对顶面/底面分别分析） |
| 维度 | 2D 克里金（与 RBF 一致） |
| 变异函数 | 默认 `spherical`，前端可选 |
| 网格分辨率 | 与 RBF 一致（`GRID_RESOLUTION`） |
| 可视化 | 静态 PNG，2x2 并排对比图 |
| 展示位置 | 右侧结果区 |
| 控制点少 | 警告后继续 |
| 缓存 | 与建模图共用缓存机制 |
| 超时 | 耐心等待（有日志轮询） |

### 1.1.3 关于"两个区域同步"的技术说明

采用**方案 A**：用 matplotlib 生成一张 2x2 并排图，四个子图在同一画布内，**天然同步**。用户看到的就是一张图，整体缩放/平移。

```python
def run_kriging_2d(x_ctrl, y_ctrl, z_ctrl, grid_res, variogram_model='spherical'):
    """
    执行 2D 普通克里金插值。
    
    参数:
        x_ctrl, y_ctrl: 控制点平面坐标（X=东, Y=北）
        z_ctrl: 控制点的高程值
        grid_res: 网格分辨率
        variogram_model: 变异函数模型
    
    返回:
        {
            'grid_x': 1D 数组,
            'grid_y': 1D 数组,
            'z_pred': 2D 数组（预测值）,
            'z_var': 2D 数组（预测方差）,
            'success': bool,
            'error': str 或 None
        }
    """
    import numpy as np
    
    n_points = len(x_ctrl)
    if n_points < 3:
        return {
            'success': False,
            'error': f'控制点数量不足（{n_points} < 3），无法插值'
        }
    
    # 去重（克里金对重复点敏感）
    points = list(zip(x_ctrl, y_ctrl, z_ctrl))
    unique_points = list(set(points))
    
    if len(unique_points) < 3:
        return {
            'success': False,
            'error': f'去重后控制点数量不足（{len(unique_points)} < 3）'
        }
    
    x_u = np.array([p[0] for p in unique_points])
    y_u = np.array([p[1] for p in unique_points])
    z_u = np.array([p[2] for p in unique_points])
    
    gx = np.linspace(x_u.min(), x_u.max(), grid_res)
    gy = np.linspace(y_u.min(), y_u.max(), grid_res)
    
    try:
        ok = OrdinaryKriging(
            x_u, y_u, z_u,
            variogram_model=variogram_model,
            nlags=6,
            enable_plotting=False,
            verbose=False
        )
        z_pred, z_var = ok.execute('grid', gx, gy)
    except Exception as e:
        return {
            'success': False,
            'error': f'克里金插值失败: {str(e)}'
        }
    
    return {
        'grid_x': gx,
        'grid_y': gy,
        'z_pred': z_pred,
        'z_var': z_var,
        'success': True,
        'error': None
    }
```

```python
def run_rbf_2d_for_comparison(x_ctrl, y_ctrl, z_ctrl, grid_x, grid_y,
                                rbf_function=None, rbf_smooth=None):
    """
    用 RBF 对同样的控制点插值，输出与克里金相同形状的网格。
    """
    import numpy as np
    from scipy.interpolate import Rbf
    from config import RBF_FUNCTION, RBF_SMOOTH
    
    rbf_function = rbf_function or RBF_FUNCTION
    rbf_smooth = rbf_smooth if rbf_smooth is not None else RBF_SMOOTH
    
    try:
        rbf = Rbf(x_ctrl, y_ctrl, z_ctrl,
                  function=rbf_function, smooth=rbf_smooth)
        gx, gy = np.meshgrid(grid_x, grid_y)
        z_pred = rbf(gx, gy)
        return z_pred
    except Exception as e:
        print(f"!!! RBF 插值失败: {e}")
        return None
```

```python
def plot_kriging_comparison(
    layer_name, part_name,
    grid_x, grid_y, z_kriging, z_variance, z_rbf,
    output_path
):
    """
    生成克里金 vs RBF 的并排对比图（2x2 布局）。
    """
    import matplotlib.pyplot as plt
    import numpy as np
    from config import FIGURE_DPI, KRIGING_VALUE_CMAP, KRIGING_VARIANCE_CMAP
    
    fig, axes = plt.subplots(2, 2, figsize=(14, 12))
    gx, gy = np.meshgrid(grid_x, grid_y)
    
    all_values = np.concatenate([
        z_rbf[~np.isnan(z_rbf)].ravel(),
        z_kriging[~np.isnan(z_kriging)].ravel()
    ])
    vmin = np.min(all_values)
    vmax = np.max(all_values)
    
    # 左上：RBF
    ax1 = axes[0, 0]
    im1 = ax1.pcolormesh(gx, gy, z_rbf, cmap=KRIGING_VALUE_CMAP,
                          vmin=vmin, vmax=vmax, shading='auto')
    ax1.set_title(f'RBF 插值 - {layer_name} {part_name}', fontsize=13, fontweight='bold')
    ax1.set_xlabel('Easting (m)')
    ax1.set_ylabel('Northing (m)')
    plt.colorbar(im1, ax=ax1, label='高程 (m)')
    
    # 右上：克里金
    ax2 = axes[0, 1]
    im2 = ax2.pcolormesh(gx, gy, z_kriging, cmap=KRIGING_VALUE_CMAP,
                          vmin=vmin, vmax=vmax, shading='auto')
    ax2.set_title(f'克里金插值 - {layer_name} {part_name}', fontsize=13, fontweight='bold')
    ax2.set_xlabel('Easting (m)')
    ax2.set_ylabel('Northing (m)')
    plt.colorbar(im2, ax=ax2, label='高程 (m)')
    
    # 左下：差异图
    ax3 = axes[1, 0]
    diff = z_kriging - z_rbf
    im3 = ax3.pcolormesh(gx, gy, diff, cmap='RdBu_r',
                          vmin=-np.max(np.abs(diff)), vmax=np.max(np.abs(diff)),
                          shading='auto')
    ax3.set_title('差异图（克里金 - RBF）', fontsize=13, fontweight='bold')
    ax3.set_xlabel('Easting (m)')
    ax3.set_ylabel('Northing (m)')
    plt.colorbar(im3, ax=ax3, label='差异 (m)')
    
    # 右下：方差
    ax4 = axes[1, 1]
    im4 = ax4.pcolormesh(gx, gy, z_variance, cmap=KRIGING_VARIANCE_CMAP,
                          shading='auto')
    ax4.set_title(f'克里金预测方差 - {layer_name} {part_name}', fontsize=13, fontweight='bold')
    ax4.set_xlabel('Easting (m)')
    ax4.set_ylabel('Northing (m)')
    plt.colorbar(im4, ax=ax4, label='方差')
    
    fig.suptitle(f'{layer_name} - {part_name} 克里金数值分析', 
                 fontsize=16, fontweight='bold', y=0.995)
    
    plt.tight_layout()
    plt.savefig(output_path, dpi=FIGURE_DPI, bbox_inches='tight', facecolor='white')
    plt.close()
    
    return output_path
```

```python
def generate_kriging_analysis_for_layer(
    layer_name, top_pts, bot_pts, output_dir, base_name,
    variogram_model=None, rbf_function=None, rbf_smooth=None
):
    """
    对单个岩性层的顶面和底面执行克里金分析，输出对比图。
    """
    import os
    import numpy as np
    from config import (
        GRID_RESOLUTION, KRIGING_GRID_RES, KRIGING_DEFAULT_VARIOGRAM,
        KRIGING_MIN_POINTS_WARN
    )
    
    grid_res = KRIGING_GRID_RES or GRID_RESOLUTION
    variogram_model = variogram_model or KRIGING_DEFAULT_VARIOGRAM
    
    warnings = []
    result = {
        'layer_name': layer_name,
        'top_path': None,
        'bot_path': None,
        'top_stats': None,
        'bot_stats': None,
        'top_control_points': [],
        'bot_control_points': [],
        'top_grid_x': [],
        'top_grid_y': [],
        'bot_grid_x': [],
        'bot_grid_y': [],
        'top_z_kriging': [],
        'top_z_variance': [],
        'bot_z_kriging': [],
        'bot_z_variance': [],
        'warnings': warnings
    }
    
    # 处理顶面
    if len(top_pts) >= 3:
        if len(top_pts) < KRIGING_MIN_POINTS_WARN:
            warnings.append(f'{layer_name} 顶面控制点仅 {len(top_pts)} 个，分析结果可靠性较低')
        
        print(f"==> 分析 {layer_name} 顶面（{len(top_pts)} 个控制点）")
        kriging_result = run_kriging_2d(
            top_pts[:, 0], top_pts[:, 1], top_pts[:, 2],
            grid_res, variogram_model
        )
        
        if kriging_result['success']:
            z_rbf = run_rbf_2d_for_comparison(
                top_pts[:, 0], top_pts[:, 1], top_pts[:, 2],
                kriging_result['grid_x'], kriging_result['grid_y'],
                rbf_function, rbf_smooth
            )
            
            if z_rbf is not None:
                top_path = os.path.join(
                    output_dir,
                    f'{base_name}_kriging_{layer_name}_top.png'
                )
                plot_kriging_comparison(
                    layer_name, '顶面',
                    kriging_result['grid_x'], kriging_result['grid_y'],
                    kriging_result['z_pred'], kriging_result['z_var'],
                    z_rbf,
                    top_path
                )
                result['top_path'] = top_path
                result['top_stats'] = {
                    'n_points': int(len(top_pts)),
                    'variance_mean': float(np.nanmean(kriging_result['z_var'])),
                    'variance_max': float(np.nanmax(kriging_result['z_var'])),
                    'variance_min': float(np.nanmin(kriging_result['z_var']))
                }
                result['top_control_points'] = [
                    (float(p[0]), float(p[1]), float(p[2])) for p in top_pts
                ]
                result['top_grid_x'] = kriging_result['grid_x'].tolist()
                result['top_grid_y'] = kriging_result['grid_y'].tolist()
                result['top_z_kriging'] = kriging_result['z_pred'].ravel().tolist()
                result['top_z_variance'] = kriging_result['z_var'].ravel().tolist()
        else:
            warnings.append(f'{layer_name} 顶面克里金失败: {kriging_result["error"]}')
    else:
        warnings.append(f'{layer_name} 顶面控制点不足（{len(top_pts)} < 3），跳过')
    
    # 处理底面
    if len(bot_pts) >= 3:
        if len(bot_pts) < KRIGING_MIN_POINTS_WARN:
            warnings.append(f'{layer_name} 底面控制点仅 {len(bot_pts)} 个，分析结果可靠性较低')
        
        print(f"==> 分析 {layer_name} 底面（{len(bot_pts)} 个控制点）")
        kriging_result = run_kriging_2d(
            bot_pts[:, 0], bot_pts[:, 1], bot_pts[:, 2],
            grid_res, variogram_model
        )
        
        if kriging_result['success']:
            z_rbf = run_rbf_2d_for_comparison(
                bot_pts[:, 0], bot_pts[:, 1], bot_pts[:, 2],
                kriging_result['grid_x'], kriging_result['grid_y'],
                rbf_function, rbf_smooth
            )
            
            if z_rbf is not None:
                bot_path = os.path.join(
                    output_dir,
                    f'{base_name}_kriging_{layer_name}_bot.png'
                )
                plot_kriging_comparison(
                    layer_name, '底面',
                    kriging_result['grid_x'], kriging_result['grid_y'],
                    kriging_result['z_pred'], kriging_result['z_var'],
                    z_rbf,
                    bot_path
                )
                result['bot_path'] = bot_path
                result['bot_stats'] = {
                    'n_points': int(len(bot_pts)),
                    'variance_mean': float(np.nanmean(kriging_result['z_var'])),
                    'variance_max': float(np.nanmax(kriging_result['z_var'])),
                    'variance_min': float(np.nanmin(kriging_result['z_var']))
                }
                result['bot_control_points'] = [
                    (float(p[0]), float(p[1]), float(p[2])) for p in bot_pts
                ]
                result['bot_grid_x'] = kriging_result['grid_x'].tolist()
                result['bot_grid_y'] = kriging_result['grid_y'].tolist()
                result['bot_z_kriging'] = kriging_result['z_pred'].ravel().tolist()
                result['bot_z_variance'] = kriging_result['z_var'].ravel().tolist()
        else:
            warnings.append(f'{layer_name} 底面克里金失败: {kriging_result["error"]}')
    else:
        warnings.append(f'{layer_name} 底面控制点不足（{len(bot_pts)} < 3），跳过')
    
    return result
```

```python
"""
分析数据管理器：保存 RBF 和克里金的分析数据，用于数据说明页面。
"""
import os
import json
import numpy as np
from datetime import datetime

from config import OUTPUT_FOLDER, CACHE_DIR_NAME


def get_analysis_data_path(session_id):
    """获取分析数据文件路径"""
    cache_dir = os.path.join(OUTPUT_FOLDER, session_id, CACHE_DIR_NAME)
    os.makedirs(cache_dir, exist_ok=True)
    return os.path.join(cache_dir, 'analysis_data.json')


def load_analysis_data(session_id):
    """加载分析数据"""
    path = get_analysis_data_path(session_id)
    if not os.path.exists(path):
        return {}
    try:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        print(f"!!! 读取分析数据失败: {e}")
        return {}


def save_analysis_data(session_id, data):
    """保存分析数据"""
    path = get_analysis_data_path(session_id)
    try:
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        return True
    except Exception as e:
        print(f"!!! 保存分析数据失败: {e}")
        return False


def _numpy_to_list(arr):
    """把 numpy 数组转为可 JSON 序列化的列表"""
    if arr is None:
        return None
    if isinstance(arr, np.ndarray):
        return [None if np.isnan(v) else float(v) for v in arr.ravel()]
    return arr


def record_rbf_data(session_id, params, layers_info):
    """记录 RBF 建模的分析数据"""
    data = load_analysis_data(session_id)
    
    if 'rbf' not in data:
        data['rbf'] = {'params': params, 'layers': [], 'created_at': None}
    
    data['rbf']['params'] = params
    data['rbf']['created_at'] = datetime.now().isoformat()
    data['rbf']['layers'] = layers_info
    
    save_analysis_data(session_id, data)
    print(f"==> RBF 分析数据已记录（{len(layers_info)} 个图层）")


def record_kriging_data(session_id, params, layers_info):
    """记录克里金分析数据"""
    data = load_analysis_data(session_id)
    
    if 'kriging' not in data:
        data['kriging'] = {'params': params, 'layers': [], 'created_at': None}
    
    data['kriging']['params'] = params
    data['kriging']['created_at'] = datetime.now().isoformat()
    data['kriging']['layers'] = layers_info
    
    save_analysis_data(session_id, data)
    print(f"==> 克里金分析数据已记录（{len(layers_info)} 个图层）")
```

```python
}
```

```javascript
async function loadAnalysisReport() {
    const reportSection = document.getElementById('reportSection');
    if (!reportSection) return;
    
    if (!sessionId) {
        reportSection.innerHTML = '<p style="text-align:center;padding:40px;color:#999;">请先上传文件</p>';
        return;
    }
    
    reportSection.innerHTML = '<p style="text-align:center;padding:40px;color:#666;">正在加载分析数据...</p>';

    } catch (error) {

function drawScatterChart(data) {
    const rbfLayer = data.data.rbf.layers.find(l => l.part_name === '顶面');
    const krigingLayer = data.data.kriging.layers.find(l => l.part_name === '顶面');
    if (!rbfLayer || !krigingLayer) return;
    
    const rbfZ = rbfLayer.z_pred.filter(v => v !== null);
    const krigingZ = krigingLayer.z_kriging.filter(v => v !== null);
    const n = Math.min(rbfZ.length, krigingZ.length, 500);
    
    const trace = {
        x: rbfZ.slice(0, n),
        y: krigingZ.slice(0, n),
        mode: 'markers',
        type: 'scatter',
        marker: { color: '#1565C0', size: 4, opacity: 0.6 },
        name: '样本'
    };
    
    const allValues = [...rbfZ.slice(0, n), ...krigingZ.slice(0, n)];
    const minV = Math.min(...allValues);
    const maxV = Math.max(...allValues);
    const diagonal = {
        x: [minV, maxV],
        y: [minV, maxV],
        mode: 'lines',
        type: 'scatter',
        line: { color: '#c62828', dash: 'dash', width: 2 },
        name: '理想对角线'
    };
    
    const layout = {
        xaxis: { title: 'RBF 插值值 (m)' },
        yaxis: { title: '克里金插值值 (m)' },
        margin: { l: 60, r: 20, t: 20, b: 60 }
    };
    
    Plotly.newPlot('scatterChart', [trace, diagonal], layout, { responsive: true });
}

function drawDiffHistogram(data) {
    const rbfLayer = data.data.rbf.layers.find(l => l.part_name === '顶面');
    const krigingLayer = data.data.kriging.layers.find(l => l.part_name === '顶面');
    if (!rbfLayer || !krigingLayer) return;
    
    const diff = [];
    const rbfZ = rbfLayer.z_pred;
    const krigingZ = krigingLayer.z_kriging;
    const n = Math.min(rbfZ.length, krigingZ.length);
    
    for (let i = 0; i < n; i++) {
        if (rbfZ[i] !== null && krigingZ[i] !== null) {
            diff.push(krigingZ[i] - rbfZ[i]);
        }
    }
    
    const trace = {
        x: diff,
        type: 'histogram',
        marker: { color: '#F57C00' },
        name: '差异'
    };
    
    const layout = {
        xaxis: { title: '克里金 - RBF (m)' },
        yaxis: { title: '频次' },
        margin: { l: 60, r: 20, t: 20, b: 60 }
    };
    
    Plotly.newPlot('diffHistogram', [trace], layout, { responsive: true });
}

function drawVarianceHistogram(data) {
    const krigingLayer = data.data.kriging.layers.find(l => l.part_name === '顶面');
    if (!krigingLayer || !krigingLayer.z_variance) return;
    
    const variance = krigingLayer.z_variance.filter(v => v !== null);
    
    const trace = {
        x: variance,
        type: 'histogram',
        marker: { color: '#2E7D32' },
        name: '方差'
    };
    
    const layout = {
        xaxis: { title: '预测方差' },
        yaxis: { title: '频次' },
        margin: { l: 60, r: 20, t: 20, b: 60 }
    };
    
    Plotly.newPlot('varianceHistogram', [trace], layout, { responsive: true });
}

async function exportAnalysisData() {
    if (!sessionId) {
        alert('请先上传文件');
        return;
    }
    
    setStatus('正在导出数据...', 'loading');
    
    try {
        const response = await fetch('/api/export_analysis', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ session_id: sessionId, formats: ['csv', 'excel'] })
        });
        
        if (!response.ok) {
            const err = await response.json();
            setStatus('!!! 导出失败: ' + (err.error || '未知错误'), 'error');
            return;
        }
        
        const blob = await response.blob();
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = `analysis_data_${Date.now()}.zip`;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(link.href);
        
        setStatus('==> 数据已导出', 'success');
    } catch (error) {
        setStatus('!!! 导出失败: ' + error.message, 'error');
    }
}
```

## 3.3 注意事项

1. **`pykrige` 的性能**：克里金比 RBF 慢 5~10 倍。如果图层多、控制点多，分析可能耗时 1~3 分钟，请耐心等待。

2. **数据体积**：网格数据可能较大（每层 30×30 = 900 个数）。10 个岩性层 × 2（顶底）× 2（RBF+Kriging）≈ 36000 个数，约 500KB。

3. **JSON 序列化**：numpy 数组需要转为 list，NaN 需要转为 null（`_numpy_to_list` 函数处理）。

4. **数据记录失败不影响主流程**：所有记录操作用 try-except 包裹。

5. **散点图取样**：如果网格点过多，最多显示 500 个点。

6. **Plotly 的引入**：用 CDN 引入，不增加 pip 依赖。

7. **文件命名**：ZIP 内部的文件名用中文，方便用户识别。

8. **Tab 切换性能**：切换"数据说明"Tab 时才加载数据，不预加载。

9. **缓存集成**：克里金使用 `task_type='kriging'`，与建模图、AI 图独立缓存。

10. **参数集成**：从 `load_session_params` 读取用户的 RBF 参数，保证对比时 RBF 使用与主建模相同的参数。
