# 多视角 AI 图改进方案：遮罩合成 + 色调统一

> **文档版本**: v2.0  
> **用途**: 供 Claude CLI 阅读，指导代码实施  
> **前置条件**: 已有可运行的钻孔三维建模工程  
> **改动范围**: processors.py + app.py + index.html + main.js + style.css

---

## 一、问题背景

当前 `/api/multiview_ai` 的实现，通过"以主锚派生"方式生成多视角 AI 图，导致：
- 3 个视角的图**形状几乎相同**（AI 把主锚当模板）
- 颜色几乎相同（AI 复用了主锚的色调）
- 视角错误（AI 没有能力推算视角关系）

**根本原因**：AI 图像生成模型没有 3D 意识，无法从 2D 图推算其他视角。

---

## 二、新方案核心思想

> **几何由 Python 100% 控制，AI 只贡献纹理。**

AI 生成的几何偏差**不影响最终结果**，因为最终用 Python 几何图的轮廓作为遮罩，只取 AI 的纹理。

def plot_geometry_with_black_edges(volume_data, view_config, output_dir, base_name):
    """
    生成带黑色描边的几何图，用于 AI 纹理升级的输入。
    黑色描边帮助 AI 识别层位边界。
    
    参数:
        volume_data: 地质体数据
        view_config: 视角配置
        output_dir: 输出目录
        base_name: 文件名前缀
    
    返回:
        保存的图片路径
    """
    import matplotlib.pyplot as plt
    
    fig = plt.figure(figsize=(FIGURE_WIDTH, FIGURE_HEIGHT))
    ax = fig.add_subplot(111, projection='3d')
    
    # 绘制地质体（带黑色描边）
    for lith, data in volume_data.items():
        color_rgb = data['color'][:3]
        layers = data['layers']
        if not layers:
            continue
        for k, Z in enumerate(layers):
            opacity = 0.2 + 0.5 * (k / (len(layers) - 1)) if len(layers) > 1 else 0.5
            ax.plot_surface(
                data['X'], data['Y'], Z,
                color=color_rgb,
                alpha=opacity,
                edgecolor='black',           # ← 关键：黑色描边
                linewidth=GEOMETRY_EDGE_WIDTH
            )
    
    # 隐藏 X/Y 轴，保留 Z 轴
    ax.set_xticks([])
    ax.set_yticks([])
    ax.set_xlabel('')
    ax.set_ylabel('')
    ax.tick_params(axis='x', which='both', length=0, labelbottom=False)
    ax.tick_params(axis='y', which='both', length=0, labelleft=False)
    ax.xaxis.line.set_visible(False)
    ax.yaxis.line.set_visible(False)
    ax.xaxis.pane.set_visible(False)
    ax.yaxis.pane.set_visible(False)
    ax.zaxis.pane.set_visible(False)
    ax.set_zlabel('Elevation (m)', fontsize=10)
    ax.tick_params(axis='z', labelsize=8)
    ax.zaxis.line.set_visible(True)
    ax.grid(False)
    
    # 范围（从 volume_data 计算）
    all_x, all_y, all_z = [], [], []
    for data in volume_data.values():
        all_x.extend(data['X'].ravel())
        all_y.extend(data['Y'].ravel())
        for layer in data['layers']:
            valid = layer[~np.isnan(layer)]
            all_z.extend(valid.tolist())
    
    if all_x and all_y and all_z:
        margin = 20
        ax.set_xlim(min(all_x) - margin, max(all_x) + margin)
        ax.set_ylim(min(all_y) - margin, max(all_y) + margin)
        ax.set_zlim(min(all_z) - 10, max(all_z) + 10)
    
    # 视角
    ax.view_init(elev=view_config['elev'], azim=view_config['azim'])
    
    # 保存
    order = view_config.get('order', 0)
    fname = f'{base_name}_geometry_{order:02d}_{view_config["name"]}.png'
    save_path = os.path.join(output_dir, fname)
    plt.savefig(save_path, dpi=FIGURE_DPI, pad_inches=0, facecolor='white')
    plt.close()
    
    return save_path
```

def generate_ai_texture_for_single_view(geometry_path, color_map, api_key=None, api_url=None, model=None):
    """
    对单张几何图调用 AI 生成纹理图。
    
    参数:
        geometry_path: 带黑色描边的几何图路径
        color_map: 颜色-岩性映射
        api_key / api_url / model: 可选覆盖
    
    返回:
        (ai_image_url, error)
    """
    import base64
    import requests
    from config import QWEN_API_KEY, QWEN_API_URL, QWEN_MODEL
    
    api_key = api_key or QWEN_API_KEY
    api_url = api_url or QWEN_API_URL
    model = model or QWEN_MODEL
    
    if not api_key:
        return None, "!!! API密钥未配置"
    
    if not os.path.exists(geometry_path):
        return None, f"!!! 几何图不存在: {geometry_path}"
    
    # 构建颜色描述
    color_lines = []
    for color, names in color_map.items():
        color_lines.append(f'- {color}: {names}')
    color_text = '\n'.join(color_lines)
    
    # 提示词：只升级纹理，不动几何
    prompt = f"""
请对参考图进行"纹理升级"——保持几何形态完全不变，只把彩色平面升级为真实的岩石纹理。

【参考图说明】
- 参考图是三维地质体的几何约束图，每一层用不同颜色表示，层位边界有黑色描边。
- 你的任务是：**保持每一层的位置、厚度、边界完全不变**，只把"彩色平面"替换为"对应岩性的真实岩石纹理"。

【岩性颜色对应关系】
{color_text}

【纹理升级要求】
1. **几何严格保持不变**：每一层的位置、厚度、边界轮廓必须与参考图完全一致。
2. **纹理逼真**：不同岩性应呈现符合地质学常识的岩石质感：
   - 透明矿物（萤石、石英）：晶莹剔透、半透明、玻璃光泽
   - 金属矿物（黄铜矿、黄铁矿、铅锌矿）：金属光泽、致密块状
   - 结晶岩（花岗斑岩、片麻岩）：斑状结构或片理构造
   - 沉积岩（砂岩、泥岩、砾岩）：颗粒感或层理清晰
   - 覆盖层（第四系、浮土）：松散、土状、无明显结晶
3. **颜色保持**：每一层的颜色基调与参考图一致，不要大幅偏移。
4. **光影自然**：岩石表面的自然光照，避免过度反光或卡通化。
5. **黑色描边保留**：参考图中的黑色层位边界必须保留，用于后续合成。
6. **背景纯白**：背景必须为纯白色（255,255,255）。

【重要提醒】
这是"纹理升级"任务，不是"重新绘制"任务。几何、视角、层位、厚度必须严格保持。
"""
    
    # 读取图片
    with open(geometry_path, 'rb') as f:
        b64 = base64.b64encode(f.read()).decode('utf-8')
    img_data = f"data:image/png;base64,{b64}"
    
    payload = {
        "model": model,
        "input": {
            "messages": [{
                "role": "user",
                "content": [
                    {"image": img_data},
                    {"text": prompt}
                ]
            }]
        },
        "parameters": {
            "n": 1,
            "size": "1024*1024",
            "watermark": False,
            "prompt_extend": True
        }
    }
    
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }
    
    try:
        response = requests.post(api_url, headers=headers, json=payload, timeout=180)
        if response.status_code != 200:
            return None, f"!!! API调用失败 ({response.status_code}): {response.text}"
        
        result = response.json()
        output = result.get('output', {})
        choices = output.get('choices', [])
        if not choices:
            return None, f"!!! API返回格式异常: {result}"
        
        content = choices[0].get('message', {}).get('content', [])
        for item in content:
            if isinstance(item, dict) and 'image' in item:
                return item['image'], None
        
        return None, f"!!! 未找到图片: {result}"
    except Exception as e:
        return None, f"!!! API调用异常: {str(e)}"
```

def composite_ai_texture_with_geometry(geo_path, ai_path, output_path):
    """
    遮罩合成：用几何图的轮廓作为遮罩，把 AI 纹理填充进去。
    
    参数:
        geo_path: Python 几何图路径（带黑色描边）
        ai_path: AI 纹理图路径
        output_path: 输出路径
    
    返回:
        输出路径
    """
    import numpy as np
    from PIL import Image
    from scipy.ndimage import binary_erosion
    from config import MASK_WHITE_THRESHOLD
    
    # 读取几何图
    geo = np.array(Image.open(geo_path).convert('RGB'))
    h, w = geo.shape[:2]
    
    # 读取 AI 图，缩放到几何图尺寸
    ai = np.array(Image.open(ai_path).convert('RGB').resize((w, h)))
    
    # 几何图中非白像素 = 地质体区域
    geo_mask = np.any(geo < MASK_WHITE_THRESHOLD, axis=2)
    
    # 找出几何图的黑色描边（边缘）
    eroded = binary_erosion(geo_mask, iterations=1)
    edge_mask = geo_mask & ~eroded
    
    # 结果：白底
    result = np.full_like(geo, 255)
    
    # 用 AI 纹理填充地质体区域
    result[geo_mask] = ai[geo_mask]
    
    # 保留几何图的黑色描边
    result[edge_mask] = geo[edge_mask]
    
    Image.fromarray(result).save(output_path)
    return output_path
```

def reinhard_color_transfer(source_img, target_img):
    """
    Reinhard 色彩迁移：把 target 的色调迁移到 source 的风格。
    
    参数:
        source_img: 参考图（numpy 数组，RGB）
        target_img: 待迁移图（numpy 数组，RGB）
    
    返回:
        迁移后的图片（numpy 数组）
    """
    import numpy as np
    from skimage import color
    
    # 转到 Lab 色彩空间
    src_lab = color.rgb2lab(source_img / 255.0)
    tgt_lab = color.rgb2lab(target_img / 255.0)
    
    # 计算均值和标准差
    src_mean = src_lab.reshape(-1, 3).mean(axis=0)
    src_std = src_lab.reshape(-1, 3).std(axis=0)
    tgt_mean = tgt_lab.reshape(-1, 3).mean(axis=0)
    tgt_std = tgt_lab.reshape(-1, 3).std(axis=0)
    
    # 避免除零
    tgt_std = np.where(tgt_std < 1e-6, 1e-6, tgt_std)
    
    # 迁移
    result_lab = (tgt_lab - tgt_mean) * (src_std / tgt_std) + src_mean
    result_rgb = color.lab2rgb(result_lab) * 255
    return result_rgb.astype(np.uint8)


def unify_multiview_color_tones(image_paths):
    """
    统一多张图的色调，以第一张为参考。
    
    参数:
        image_paths: 图片路径列表（第一张作为参考）
    
    返回:
        修改后的图片路径列表
    """
    import numpy as np
    from PIL import Image
    from config import MULTIVIEW_COLOR_METHOD
    
    if len(image_paths) < 2:
        return image_paths
    
    # 读取第一张作为参考
    ref_img = np.array(Image.open(image_paths[0]).convert('RGB'))
    
    result_paths = [image_paths[0]]  # 第一张不变
    
    for i, path in enumerate(image_paths[1:], start=1):
        tgt_img = np.array(Image.open(path).convert('RGB'))
        
        # 尺寸对齐
        if tgt_img.shape != ref_img.shape:
            tgt_img = np.array(Image.open(path).convert('RGB').resize(
                (ref_img.shape[1], ref_img.shape[0])
            ))
        
        # 应用色彩迁移
        if MULTIVIEW_COLOR_METHOD == 'reinhard':
            result_img = reinhard_color_transfer(ref_img, tgt_img)
        else:
            # 简单的直方图匹配（备选）
            result_img = tgt_img  # 暂不实现
        
        # 保存
        out_path = path.replace('.png', '_unified.png')
        Image.fromarray(result_img).save(out_path)
        result_paths.append(out_path)
    
    return result_paths

@app.route('/api/multiview_texture_upgrade', methods=['POST'])
def multiview_texture_upgrade_api():
    """
    多视角纹理升级：
    1. Python 生成 3 张带黑色描边的几何图
    2. 每张独立调用 AI 升级纹理
    3. 遮罩合成
    4. 色调统一
    """
    import requests
    
    data = request.json
    session_id = data.get('session_id')
    view_count = data.get('view_count', 3)
    color_map = data.get('color_map', {})
    
    if not session_id:
        return jsonify({'error': '缺少 session_id'}), 400
    
    if not color_map:
        return jsonify({'error': '缺少颜色映射，请先生成建模图'}), 400
    
    try:
        from config import MULTIVIEW_VIEW_ORDERS, VIEW_CONFIGS, OUTPUT_FOLDER
        
        # 从定位表读取数据，重新生成 volume_data（如果没缓存）
        # 这里假设已有的 volume_data 可以通过缓存获取
        # 如果没有缓存，需要重新读取文件
        
        # ... 加载 loc_df, survey_df, litho_df ...
        # ... 构建 volume_data ...
        
        # Step 1: 生成几何图
        geometry_paths = []
        for order in MULTIVIEW_VIEW_ORDERS[:view_count]:
            view = next((v for v in VIEW_CONFIGS if v.get('order') == order), None)
            if not view:
                continue
            
            geo_path = plot_geometry_with_black_edges(
                volume_data, view, OUTPUT_FOLDER, f'multiview_{session_id}'
            )
            geometry_paths.append((order, view, geo_path))
        
        # Step 2: 每张独立调用 AI
        ai_paths = []
        success_count = 0
        
        for order, view, geo_path in geometry_paths:
            log_and_print(session_id, f"==> 正在生成 图{order} ({view['label']}) 的 AI 纹理")
            
            ai_url, error = generate_ai_texture_for_single_view(geo_path, color_map)
            
            if error or not ai_url:
                log_and_print(session_id, f"!!! 图{order} AI 生成失败: {error}")
                continue
            
            # 下载 AI 图
            local_filename = f'multiview_ai_raw_{order:02d}_{view["name"]}.png'
            local_path = os.path.join(OUTPUT_FOLDER, local_filename)
            
            resp = requests.get(ai_url, timeout=60)
            if resp.status_code == 200:
                with open(local_path, 'wb') as f:
                    f.write(resp.content)
                ai_paths.append((order, view, geo_path, local_path))
                success_count += 1
        
        if success_count == 0:
            return jsonify({'error': '所有视角 AI 生成失败，请重试'}), 500
        
        # Step 3: 遮罩合成
        composite_paths = []
        for order, view, geo_path, ai_path in ai_paths:
            out_filename = f'multiview_composite_{order:02d}_{view["name"]}.png'
            out_path = os.path.join(OUTPUT_FOLDER, out_filename)
            composite_ai_texture_with_geometry(geo_path, ai_path, out_path)
            composite_paths.append(out_path)
        
        # Step 4: 色调统一
        unified_paths = unify_multiview_color_tones(composite_paths)
        
        # 组织返回数据
        images = []
        for i, (order, view, _, _) in enumerate(ai_paths):
            images.append({
                'view_name': view['name'],
                'view_label': view['label'],
                'order': order,
                'url': f'/outputs/{os.path.basename(unified_paths[i])}',
                'has_ai': True
            })
        
        return jsonify({
            'success': True,
            'images': images,
            'success_count': success_count,
            'total_count': len(geometry_paths)
        })
    
    except Exception as e:
        log_and_print(session_id, f"!!! 多视角纹理升级失败: {e}")
        import traceback
        traceback.print_exc()
        return jsonify({'error': str(e)}), 500
```

## 十一、注意事项

1. **AI 生成的几何偏差不影响最终结果**：因为最终用 Python 几何图作为遮罩，AI 只贡献纹理。

2. **色调统一以第一张为参考**：图1 是主锚，图3、图5 对齐到图1。

3. **如果 AI 生成的纹理效果不好**：可以调整提示词，或在 `config.py` 中增加"重试"逻辑。

4. **几何图带黑色描边的作用**：帮助 AI 识别层位边界，同时合成时保留描边。

5. **scikit-image 是必需依赖**：用于 Reinhard 色彩迁移（Lab 色彩空间转换）。

6. **如果 Reinhard 效果不理想**：可以在 `config.py` 中切换为 `'histogram'`（需另行实现），或关闭色调统一。
